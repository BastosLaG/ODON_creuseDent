87 public void SetPipelineShaders()
88 {
89 if (m_ShaderContainers == null)
90 return;
91
92 bool isBuiltinRenderPipeline = GraphicsSettings.currentRenderPipeline == null;
93
94 foreach (var info in m_ShaderContainers)
95 {
96 if (info.material == null)
97 continue;
98
99 // Find the appropriate shaders based on the toggle
100 Shader birpShader = info.useBuiltinShaderName ? Shader.Find(info.builtInPipelineShaderName) : info.builtInPipelineShader;
101 Shader srpShader = info.useSRPShaderName ? Shader.Find(info.scriptableRenderPipelineShaderName) : info.scriptableRenderPipelineShader;
102
103 // Determine current shader for comparison
104 Shader currentShader = info.material.shader;
105
106 // Update shader for the current render pipeline only if necessary
107 if (isBuiltinRenderPipeline && birpShader != null && currentShader != birpShader)
108 {
109 info.material.shader = birpShader;
110 MarkMaterialModified(info.material);
111 }
112 else if (!isBuiltinRenderPipeline && srpShader != null && currentShader != srpShader)
113 {
114 info.material.shader = srpShader;
115 MarkMaterialModified(info.material);
116 }
117 }
118 }

199 public override float GetPropertyHeight(SerializedProperty property, GUIContent label)
200 {
201 const int baseFieldCount = 4; // The Material field, the two toggles, and one for an optional field.
202 int extraLineCount = property.FindPropertyRelative("useBuiltinShaderName").boolValue ? 0 : 1;
203 extraLineCount += property.FindPropertyRelative("useSRPShaderName").boolValue ? 0 : 1;
204
205 float singleLineHeight = EditorGUIUtility.singleLineHeight;
206 float verticalSpacing = EditorGUIUtility.standardVerticalSpacing;
207 float headerHeight = EditorGUIUtility.singleLineHeight; // No longer need extra height for headers.
208
209 // Calculate height for fields and headers
210 float fieldsHeight = baseFieldCount * singleLineHeight + (baseFieldCount - 1 + extraLineCount) * verticalSpacing;
211
212 // Allow space for header, separator line, and a bit of padding before the line.
213 float headersHeight = 2 * (headerHeight + verticalSpacing);
214 float separatorSpace = verticalSpacing / 2 + 1; // Additional vertical spacing and line height.
215
216 return fieldsHeight + headersHeight + separatorSpace + singleLineHeight * 1.5f;
217 }
