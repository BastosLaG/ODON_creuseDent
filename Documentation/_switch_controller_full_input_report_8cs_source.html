51 var leftStickVec = Vector2.zero;

53 if (controllerType == SpecificControllerTypeEnum.LeftJoyCon ||

62 var rawLeftStickVert = (l1 >> 4) | (l2 << 4);
63
64 var leftStickX = (Mathf.InverseLerp(lStickCalibData.xMin, lStickCalibData.xMax, rawLeftStickHoriz) * 2) - 1;
65 var leftStickY = (Mathf.InverseLerp(lStickCalibData.yMin, lStickCalibData.yMax, rawLeftStickVert) * 2) - 1;
66 leftStickVec = new Vector2(leftStickX, leftStickY);
67
68 // Debug.Log($"Left stick data: Raw: ({rawLeftStickHoriz:X3};{rawLeftStickVert:X3}) Calibration data: Center=({lStickCalibData.xCenter:X3},{lStickCalibData.yCenter:X3}); X=[{lStickCalibData.xMin:X3} - {lStickCalibData.xMax:X3}]; Y=[{lStickCalibData.yMin:X3} - {lStickCalibData.yMax:X3}] Final data: {leftStickVec}");
69 }
70

84 rightStickVec = new Vector2(rightStickX, rightStickY);
85
86 // Debug.Log($"Right stick data: Raw: ({rawRightStickHoriz:X3};{rawRightStickVert:X3}) Calibration data: Center=({rStickCalibData.xCenter:X3},{rStickCalibData.yCenter:X3}); X=[{rStickCalibData.xMin:X3} - {rStickCalibData.xMax:X3}]; Y=[{rStickCalibData.yMin:X3} - {rStickCalibData.yMax:X3}] Final data: {rightStickVec}");

104

114 orientation = currentOrientation + tempGyro,
115 };

118 state.Set(SwitchControllerVirtualInputState.Button.X, (rightButtons & 0x02) != 0);

123 state.Set(SwitchControllerVirtualInputState.Button.R, (rightButtons & 0x40) != 0);
124 state.Set(SwitchControllerVirtualInputState.Button.ZR, (rightButtons & 0x80) != 0);
125 state.Set(SwitchControllerVirtualInputState.Button.Minus, (sharedButtons & 0x01) != 0);
