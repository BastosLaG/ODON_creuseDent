91 public static void Initialize() {
92 scormAPIWrapper = new ScormAPIWrapper(objectName,"ScormValueCallback");
93 scormAPIWrapper.Initialize();
94 try {
95 if(scormAPIWrapper.IsScorm2004)
96 {
97 //Load StudentRecord data using SCORM 2004
98 studentRecord = LoadStudentRecord();
99 } else {
100 //Load StudentRecord data using SCORM 1.2
101 throw new System.InvalidOperationException("SCORM 1.2 not currently supported"); //TODO: Not currently supported.
102 }
103 } catch(Exception e) {
104 wgldebugPrint("***Initialize***" + e.Message + "<br/>" + e.StackTrace + "<br/>" + e.Source);
105 }
106 initialized = true;
107 GameObject.Find(objectName).BroadcastMessage("Scorm_Initialize_Complete",SendMessageOptions.DontRequireReceiver);
108 }

453 try {
454 string identifier;
455 string strValue;
456
457 interaction.id = "urn:STALS:interaction-id-" + studentRecord.interactions.Count.ToString (); //Override ID to ensure it is unique

472 scormAPIWrapper.SetValue(identifier,strValue);
473
474 identifier = "cmi.interactions."+i+".timestamp";
475 strValue = String.Format("{0:s}", interaction.timeStamp);
476 scormAPIWrapper.SetValue(identifier,strValue);
477
478 identifier = "cmi.interactions."+i+".weighting";
479 strValue = interaction.weighting.ToString();
480 scormAPIWrapper.SetValue(identifier,strValue);
481
482 identifier = "cmi.interactions."+i+".learner_response";
483 strValue = interaction.response;
484 scormAPIWrapper.SetValue(identifier,strValue);
485
486 identifier = "cmi.interactions."+i+".result";
487 strValue = CustomTypeToString (interaction.result);
488 if (interaction.result == StudentRecord.ResultType.estimate) {
489 strValue = interaction.estimate.ToString();
490 }
491 scormAPIWrapper.SetValue(identifier,strValue);
492
493 identifier = "cmi.interactions."+i+".latency";
494 strValue = secondsToTimeInterval(interaction.latency);
495 scormAPIWrapper.SetValue(identifier,strValue);
496
497 identifier = "cmi.interactions."+i+".description";
498 strValue = interaction.description;
499 scormAPIWrapper.SetValue(identifier,strValue);
500
501 if(interaction.objectives != null) {
502 int objectivesCount = interaction.objectives.Count;
503 if(objectivesCount != 0) {
504 for (int x = 0; x < objectivesCount; i++) {
505 identifier = "cmi.interactions."+i+".objectives."+x+".id";
506 strValue = interaction.objectives[x].id;
507 scormAPIWrapper.SetValue(identifier,strValue);
508 }
509 }
510 }
511
512 if(interaction.correctResponses != null) {
513 int correctResponsesCount = interaction.correctResponses.Count;
514 if(correctResponsesCount != 0) {
515 for (int x = 0; x < correctResponsesCount; i++) {
516 identifier = "cmi.interactions."+i+".correct_responses."+x+".pattrern";
517 strValue = interaction.correctResponses[x].pattern;
518 scormAPIWrapper.SetValue(identifier,strValue);
519 }
520 }
521 }
522
523
524
525
526 } catch(System.Exception e) {
527 wgldebugPrint("***AddInteraction***" + e.Message + "<br/>" + e.StackTrace + "<br/>" + e.Source);
528 }

1198 private static StudentRecord LoadStudentRecord() {
1199 studentRecord = new StudentRecord();
1200
1201 studentRecord.version = scormAPIWrapper.GetValue ("cmi._version");
1202
1203 //Comments From Learner
1204 int commentsFromLearnerCount = ParseInt (scormAPIWrapper.GetValue ("cmi.comments_from_learner._count"));
1205 studentRecord.commentsFromLearner = new List<StudentRecord.CommentsFromLearner>();
1206
1207 if (commentsFromLearnerCount != 0) {
1208
1209 for (int i = 0; i < commentsFromLearnerCount; i++) {
1210 string comment = scormAPIWrapper.GetValue ("cmi.comments_from_learner."+i+".comment");
1211 string location = scormAPIWrapper.GetValue ("cmi.comments_from_learner."+i+".location");
1212 DateTime timestamp = DateTime.Parse( scormAPIWrapper.GetValue ("cmi.comments_from_learner."+i+".timestamp") );

1619 private static float timeIntervalToSeconds(string timeInterval) {
1620 float totalSeconds = 0f;
1621
1622 if (timeInterval != "") {
1623
1624 long hundredthsPerYear = 3155760000;
1625 long hundredthsPerMonth = 262980000;
1626 long hundredthsPerDay = 8640000;
1627 long hundredthsPerHour = 360000;
1628 long hundredthsPerMinute = 6000;
1629 long hundredthsPerSecond = 100;
1630
1631 Regex re = new Regex (@"P(([0-9]+)Y)?(([0-9]+)M)?(([0-9]+)D)?T?(([0-9]+)H)?(([0-9]+)M)?(([0-9]+)(\.[0-9]+)?S)?");
1632 Match m = re.Match (timeInterval);
1633
1634 int years = int.Parse (m.Groups [2].Value == "" ? "0" : m.Groups [2].Value);
1635 int months = int.Parse (m.Groups [4].Value == "" ? "0" : m.Groups [4].Value);
1636 int days = int.Parse (m.Groups [6].Value == "" ? "0" : m.Groups [6].Value);
1637 int hours = int.Parse (m.Groups [8].Value == "" ? "0" : m.Groups [8].Value);
1638 int minutes = int.Parse (m.Groups [10].Value == "" ? "0" : m.Groups [10].Value);
1639 float seconds = float.Parse (m.Groups [12].Value == "" ? "0" : m.Groups [12].Value);
1640 float tenths = float.Parse (m.Groups [13].Value == "" ? "0" : "0" + m.Groups [13].Value);
1641 seconds = seconds + tenths;
1642
1643 long totalHundredthsOfASecond = 0; //work at the hundreths of a second level because that is all the precision that is required
1644 totalHundredthsOfASecond += years * hundredthsPerYear;
1645 totalHundredthsOfASecond += months * hundredthsPerMonth;
1646 totalHundredthsOfASecond += days * hundredthsPerDay;
1647 totalHundredthsOfASecond += hours * hundredthsPerHour;
1648 totalHundredthsOfASecond += minutes * hundredthsPerMinute;
1649 totalHundredthsOfASecond += Convert.ToInt64 (seconds * hundredthsPerSecond);
1650
1651 totalSeconds = Convert.ToSingle (totalHundredthsOfASecond) / 100;
1652 }
1653
1654 return totalSeconds;
1655 }
