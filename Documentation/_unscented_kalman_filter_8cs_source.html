192 private Matrix<double>[] UnscentedTransform(Matrix<double> X, Matrix<double> Wm, Matrix<double> Wc, int n, Matrix<double> R)
193 {
194 int L = X.ColumnCount;
195 int dim = X.RowCount;
196
197 if (n != dim)
198 throw new ArgumentException($"Mismatch: n ({n}) must match number of rows in X ({dim})");
199
200 Matrix<double> y = Matrix.Build.Dense(n, 1, 0);
201 Matrix<double> Y = Matrix.Build.Dense(n, L, 0);
202
203 for (int k = 0; k < L; k++)
204 {
205 Matrix<double> col = X.Column(k).ToColumnMatrix();
206 Y.SetSubMatrix(0, n, k, 1, col);
207 y = y.Add(col.Multiply(Wm[0, k]));
208 }
209
210 Matrix<double> Y1 = Y - y * DenseMatrix.Create(1, L, 1.0);
211 Matrix<double> P = Y1 * DenseMatrix.OfDiagonalArray(Wc.Row(0).ToArray()) * Y1.Transpose() + R;
212
213 return new Matrix<double>[] { y, Y, P, Y1 };
214 }
