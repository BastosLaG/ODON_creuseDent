256 public void AddLearnerInteractionToList (StudentRecord.LearnerInteractionRecord record) {
257 string stringInteractionsList = "";
258
259 string timeStamp = record.timeStamp.ToString(); // Get the Interaction data as strings
260 String id = record.id;
261 string type = record.type.ToString();
262 string weighting = record.weighting.ToString();
263 string latency = record.latency.ToString();
264 string description = record.description;
265 string response = record.response;
266
267 string result = record.result.ToString(); // If the Interaction result is set as the 'estimate', the result needs to be set to the numeric estimate value
268 if(record.result == StudentRecord.ResultType.estimate) {
269 result = record.estimate.ToString();
270 }
271
272 string stringInteractionObjectivesList = ""; // Load the list of objectives and format it into a string
273 if(record.objectives != null) {
274 List<StudentRecord.LearnerInteractionObjective> interactionObjectivesList = record.objectives;
275 if(interactionObjectivesList.Count > 0) {
276 string stringObjectiveIds = "";
277 foreach(StudentRecord.LearnerInteractionObjective singleObjective in interactionObjectivesList) {
278 stringObjectiveIds += singleObjective.id + " ";
279 }
280 stringInteractionObjectivesList = "Objectives ("+ stringObjectiveIds+")\n";
281 }
282 }
283
284 string stringInteractionCorrectResponseList = ""; // Load the list of correct response patterns and format it into a string
285 if(record.correctResponses != null) {
286 List<StudentRecord.LearnerInteractionCorrectResponse> interactionCorrectResponseList = record.correctResponses;
287 if(interactionCorrectResponseList.Count > 0) {
288 string stringPatterns = "";
289 foreach(StudentRecord.LearnerInteractionCorrectResponse singleCorrectResponse in interactionCorrectResponseList) {
290 stringPatterns += singleCorrectResponse.pattern + " ";
291 }
292 stringInteractionCorrectResponseList = "Correct Response Patterns ("+ stringPatterns+")\n";
293 }
294 }
295
296 stringInteractionsList += timeStamp + " :: " + id + "\n" + description + "\n(type: "+type+" weighting: "+weighting+" latency: "+latency+"secs)\nResponse: "+response+" Result: "+result+"\n"+stringInteractionCorrectResponseList+stringInteractionObjectivesList+"\n"; // Format the entire Interaction data set as a string and add to the GUI element
297 GameObject.Find ("InteractionList").GetComponent<Text> ().text += stringInteractionsList;
298 }

546 public void ButtonAddInteractionPressed() {
547 GameObject inputFieldDescription = GameObject.Find("InputFieldDescription");
548 GameObject inputFieldWeighting = GameObject.Find("InputFieldWeighting");
549 GameObject inputFieldResponse = GameObject.Find("InputFieldResponse");
550 GameObject toggleCorrect = GameObject.Find("ToggleCorrect");
551
552 string description = inputFieldDescription.GetComponent<InputField>().text;
553 string weighting = inputFieldWeighting.GetComponent<InputField>().text;
554 string studentResponse = inputFieldResponse.GetComponent<InputField>().text;
555 bool correct = toggleCorrect.GetComponent<Toggle>().isOn;
556
557 if (description == "" | weighting == "" | studentResponse == "") { // Validate the Input Elements
558 if (description == "") {
559 inputFieldDescription.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "You must enter a description!";
560 }
561 if (weighting == "") {
562 inputFieldWeighting.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "You must enter a weighting!";
563 }
564 if (studentResponse == "") {
565 inputFieldResponse.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "You must enter a student response!";
566 }
567 } else { // Add the Interaction
568 StudentRecord.LearnerInteractionRecord newInteraction = new StudentRecord.LearnerInteractionRecord ();
569 newInteraction.id = ScormManager.GetNextInteractionId();
570 newInteraction.timeStamp = DateTime.Now;
571 newInteraction.type = StudentRecord.InteractionType.other;
572 newInteraction.weighting = float.Parse(weighting);
573 newInteraction.response = studentResponse;
574 newInteraction.latency = 18.4f;
575 newInteraction.description = description;
576 StudentRecord.ResultType result = StudentRecord.ResultType.incorrect;
577 if (correct) {
578 result = StudentRecord.ResultType.correct;
579 }
580 newInteraction.result = result;
581
582 ScormManager.AddInteraction (newInteraction);
583 AddLearnerInteractionToList (newInteraction);
584
585 //Reset
586 inputFieldDescription.GetComponent<InputField>().text = "";
587 inputFieldWeighting.GetComponent<InputField>().text = "";
588 inputFieldResponse.GetComponent<InputField>().text = "";
589 toggleCorrect.GetComponent<Toggle>().isOn = true;
590 inputFieldDescription.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "Description...";
591 inputFieldWeighting.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "Weighting...";
592 inputFieldResponse.transform.Find ("Placeholder").gameObject.GetComponent<Text> ().text = "Student response...";
593 }
