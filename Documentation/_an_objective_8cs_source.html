67 public void Initialise(int newIndex, StudentRecord.Objectives newData) {
68 isInitialising = true; // Set the isInitialising flag to true (stops the updating of the Objective while the data is loading
69
70 index = newIndex;
71 data = newData;
72
73 objectiveIndexGO.GetComponent<Text> ().text = index.ToString (); // Set the text of the Objective Index GUI Element
74 objectiveIDGO.GetComponent<Text> ().text = newData.id; // Set the text of the Objective ID GUI Element
75 objectiveDescriptionGO.GetComponent<Text> ().text = newData.description; // Set the text of the Objective Description GUI Element
76
77 objectiveScoreMinGO.GetComponent<InputField> ().text = newData.score.min.ToString(); // Set the text of the Min Score GUI Element
78 objectiveScoreMaxGO.GetComponent<InputField> ().text = newData.score.max.ToString(); // Set the text of the Max Score GUI Element
79 objectiveScoreRawGO.GetComponent<InputField> ().text = newData.score.raw.ToString(); // Set the text of the Raw Score GUI Element
80
81 objectiveProgressMeasureGO.GetComponent<Slider> ().value = newData.progressMeasure; // Set the text of the Progress Measure GUI Element
82
83 switch(newData.successStatus) { // Set the correct Success Status toggle
84 case StudentRecord.SuccessStatusType.passed:
85 objectiveSuccessPassedGO.GetComponent<Toggle>().isOn = true;
86 break;
87 case StudentRecord.SuccessStatusType.failed:
88 objectiveSuccessFailedGO.GetComponent<Toggle>().isOn = true;
89 break;
90 case StudentRecord.SuccessStatusType.unknown:
91 objectiveSuccessUnknownGO.GetComponent<Toggle>().isOn = true;
92 break;
93 }
94
95 switch(newData.completionStatus) { // Set the correct Completion Status toggle
96 case StudentRecord.CompletionStatusType.completed:
97 objectiveCompletedGO.GetComponent<Toggle>().isOn = true;
98 break;
99 case StudentRecord.CompletionStatusType.incomplete:
100 objectiveIncompleteGO.GetComponent<Toggle>().isOn = true;
101 break;
102 case StudentRecord.CompletionStatusType.not_attempted:
103 objectiveNotAttemptedGO.GetComponent<Toggle>().isOn = true;
104 break;
105 case StudentRecord.CompletionStatusType.unknown:
106 objectiveUnknownGO.GetComponent<Toggle>().isOn = true;
107 break;
108 }
109
110 ResizeParent (); // I could not get the Content Size Fitter working correctly on the PanelObjectivesList GUI element, calculated manually
111 isInitialising = false; // Initialise complete, update reactivated.
112 }
113
117 private void ResizeParent() {
118 Transform parent = this.transform.parent; // The PanelObjectivesList GUI element
119
120 int count = 0; // Count the number of Objective panels
121 foreach (Transform child in parent) {
122 count++;
123 }
124
125 float height = count * 320f; // Calculate and set the height of the PanelObjectivesList GUI element
126 parent.GetComponent<RectTransform> ().sizeDelta = new Vector2 (492f, height);
127 }
