195 APICallResult WaitForReturn(int key) {
196 int timeout = 0;
197 bool wait = true;
198
199 lock (CallbackValues) {
200 wait = CallbackValues[key] == null && timeout < TimeToWaitForReply; //Wait = true if the CallbackValues for this call is empty (not yet processed or timeout is reached
201 }
202
203 while (wait) {
204 processQueue(); //Look for any queued results that match this call's key
205 lock (CallbackValues) {
206 wait = CallbackValues[key] == null && timeout < TimeToWaitForReply;
207 }
208 if(wait)
209 System.Threading.Thread.Sleep(TimePerPoll); //Sleep for the poll period (so we don't check obsesively)
210 timeout += TimePerPoll;
211 }
212
213 if (timeout >= TimeToWaitForReply) //On timeout send log call too
214 Log("timeout");
215
216 lock (CallbackValues) {
217 APICallResult ret = CallbackValues[key]; //Fetch the callback values for this call (key)
218 CallbackValues.Remove(key); //Remove this callback from the list (don't need to process it as we now have the value)
219 return ret; //Return the APICallResult for this call
220 }
221 }

231 void processQueue() {
232 try {
233 lock (ResponseQueue) {
234 while (ResponseQueue.Count > 0) { //Only process if we have something in the queue
235
236 string input;
237 input = ResponseQueue.Dequeue(); //Get the next queued response from scorm.js
238
239 string[] tokens = input.Split('|'); //Split the result into the components (see the scorm.js file for details
240 int key = System.Convert.ToInt32(tokens[tokens.Length-1]); //Fetch the key (as an int32)
241
242 lock (CallbackValues) {
243 CallbackValues[key] = new ScormAPIWrapper.APICallResult(); //Create a new APICallResult object for this call (key)
244 CallbackValues[key].Result = tokens[0]; //Fetch the result string
245 CallbackValues[key].ErrorCode = tokens[1]; //Fetch the error code
246 CallbackValues[key].ErrorDescription = tokens[2]; //Fetch the error description
247 CallbackValues[key].Key = tokens[tokens.Length-1]; //Fetch the key (as a string)
248 }
249 }
250 }
251 } catch (Exception e) {
252 wgldebugPrint("***processQueue***" + e.Message + "<br/>" + e.StackTrace + "<br/>" + e.Source);
253 }
254
255 }

310 public string GetValue(string identifier) {
311 string result = "";
312
313 int key = SetupCallback(); //Set the key for this call (identifies the call when processing the returned results in the queue
314 Log("Get " + identifier);
315
316 wgldoGetValue(identifier, CallbackObjectName, CallbackFunctionName, key);
317
318 APICallResult returnval = WaitForReturn(key);
319
320 result = returnval.Result;
321
322 if(returnval.ErrorCode == "")
323 Log("Got " + result); //No error, log result
324 else
325 Log("Error:" + returnval.ErrorCode.ToString() + " " + returnval.ErrorDescription + " Result: "+returnval.Result); //Error, log error code and description
326
327 return result;
328 }
